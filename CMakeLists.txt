cmake_minimum_required(VERSION 3.24)
project(CodexDashboard VERSION 0.1.1 LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This native desktop build targets Windows.")
endif()
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(pulse_quota STATIC src/pulse_quota.c)
target_include_directories(pulse_quota PUBLIC include)
add_executable(codex-dashboard src/sidecar_main.cpp src/codex_client.cpp src/session_stats.cpp src/cdp_client.cpp src/desktop_launch.cpp)
target_include_directories(codex-dashboard PRIVATE third_party)
target_link_libraries(codex-dashboard PRIVATE pulse_quota winhttp ole32 uuid)
target_compile_definitions(codex-dashboard PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602 NTDDI_VERSION=0x06020000)
add_executable(codex-launcher WIN32 src/desktop_start.cpp src/desktop_launch.cpp src/cdp_client.cpp)
target_include_directories(codex-launcher PRIVATE include third_party)
target_link_libraries(codex-launcher PRIVATE winhttp ole32 uuid iphlpapi ws2_32)
target_compile_definitions(codex-launcher PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602 NTDDI_VERSION=0x06020000)
if(MSVC)
  target_compile_options(pulse_quota PRIVATE /W4 /utf-8)
  target_compile_options(codex-dashboard PRIVATE /W4 /utf-8)
  target_compile_options(codex-launcher PRIVATE /W4 /utf-8)
else()
  target_compile_options(pulse_quota PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(codex-dashboard PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(codex-launcher PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(codex-dashboard PRIVATE -municode -static -static-libgcc -static-libstdc++)
  target_link_options(codex-launcher PRIVATE -municode -static -static-libgcc -static-libstdc++)
  if(CMAKE_BUILD_TYPE STREQUAL "Release")
    target_compile_options(codex-dashboard PRIVATE -Os)
    target_link_options(codex-dashboard PRIVATE -s)
    target_compile_options(codex-launcher PRIVATE -Os)
    target_link_options(codex-launcher PRIVATE -s)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(quota-tests tests/quota_tests.c)
  target_link_libraries(quota-tests PRIVATE pulse_quota)
  add_test(NAME quota-contract COMMAND quota-tests)
  add_executable(session-tests tests/session_tests.cpp src/session_stats.cpp)
  target_link_libraries(session-tests PRIVATE pulse_quota)
  if(NOT MSVC)
    target_compile_options(session-tests PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(session-tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME session-contract COMMAND session-tests)
  add_executable(reset-tests tests/reset_policy_tests.cpp src/codex_client.cpp)
  target_include_directories(reset-tests PRIVATE third_party)
  target_link_libraries(reset-tests PRIVATE pulse_quota)
  target_compile_definitions(reset-tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  if(NOT MSVC)
    target_link_options(reset-tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME reset-policy COMMAND reset-tests)
endif()
